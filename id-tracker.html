2

        
Paste This Script

      

      

        Handles writing the student ID to the selected sheet and
        automatically removes it from the other two sheets if it exists there.
        

          
          function doGet(e) {
  var p      = e && e.parameter ? e.parameter : {};
  var ss     = SpreadsheetApp.getActiveSpreadsheet();
  var target = p.sheet;
  var col    = (p.col || "A").toUpperCase();
  var colIdx = col.charCodeAt(0) - 64;
  var data   = p.data ? String(p.data).trim() : "";
  var all    = ["INVENTORY", "W/Proj", "DEPLOYED"];

  // Get target sheet, fall back to active sheet
  var sheet  = ss.getSheetByName(target) || ss.getActiveSheet();

  // Remove this ID from the OTHER two sheets
  all.forEach(function(name) {
    if (name === target) return;
    var sh = ss.getSheetByName(name);
    if (!sh) return;
    if (sh.getLastRow() < 1) return;
    var vals = sh.getDataRange().getValues();
    for (var i = vals.length - 1; i >= 0; i--) {
      for (var j = 0; j < vals[i].length; j++) {
        if (String(vals[i][j]).trim() === data) {
          sh.deleteRow(i + 1);
        }
      }
    }
  });

  // Remove empty rows from target sheet
  var maxRow = sheet.getMaxRows();
  for (var i = maxRow; i >= 1; i--) {
    var numCols = sheet.getLastColumn() || 1;
    var row = sheet.getRange(i, 1, 1, numCols).getValues()[0];
    if (row.every(function(c) { return c === "" || c === null; })) {
      sheet.deleteRow(i);
    }
  }

  // Append the ID to the next available row
  var lastRow = sheet.getLastRow();
  sheet.getRange(lastRow + 1, colIdx).setValue(data);

  return respond({ok: true, row: lastRow + 1});
}

function respond(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) { return doGet(e); }
        

      

    


    

      

        
3

        
Deploy as Web App

      

      

        Click Deploy → New deployment → Web App


        Required settings:

        · Execute as: Me

        · Who has access: Anyone


        Authorize when prompted, then copy the /exec URL and paste it in Settings.
        

          ⚠ Every time you edit the script you must create a New Version — edits don't go live automatically.

          💡 Use Test Connection in Settings after deploying to confirm everything works before scanning.
        

      

    


    

      

        
5

        
Scan

      

      

        NFC tags must contain the student ID as a plain text record (string format, e.g. 202112345).
        Select a destination tab, tap the circle, and hold a tag to the back of your Android device.
        The ID is logged locally and sent to Google Sheets.
        

          📱 Web NFC only works on Chrome on Android. It will not work on iOS or desktop.
